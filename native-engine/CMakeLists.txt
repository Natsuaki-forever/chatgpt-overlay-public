cmake_minimum_required(VERSION 3.20)
project(OverlayNative LANGUAGES CXX)
add_executable(OverlayNative WIN32 src/main.cpp)
target_compile_features(OverlayNative PRIVATE cxx_std_17)
target_compile_definitions(OverlayNative PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(OverlayNative PRIVATE user32 gdi32 dwmapi d2d1 dwrite windowscodecs ole32 shell32 advapi32 psapi)
if(MSVC)
  target_compile_options(OverlayNative PRIVATE /utf-8 /EHsc)
  set_property(TARGET OverlayNative PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
